{{- /* JSON-LD structured data. - Person on the homepage, so search engines can tie the site to an entity. sameAs mirrors the profile links in the baseof.html footer. - BlogPosting on dated pages -- the same condition baseof.html uses to decide og:type=article, kept deliberately in sync. - BreadcrumbList on pages inside a section (/poznamky/*, /manim/*, ...). Skipped entirely for `redirect:` stubs (the content is not ours) and for noindex pages (nothing should describe what we ask not to be indexed). */ -}} {{- $isRedirect := and .Params.redirect .Params.redirect.to -}} {{- if not (or $isRedirect .Params.noindex) -}} {{- if .IsHome -}} {{- end -}} {{- if and (not .IsHome) .Date (ne .Date.Unix 0) -}} {{- $post := dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" (.Title | plainify) "url" .Permalink "mainEntityOfPage" .Permalink "datePublished" (.Date.Format "2006-01-02") "inLanguage" (.Params.language | default site.Language.Lang) "author" (dict "@type" "Person" "name" site.Params.author "url" site.BaseURL) -}} {{- with .Description }}{{ $post = merge $post (dict "description" (. | plainify)) }}{{ end -}} {{- if not .Lastmod.IsZero }}{{ $post = merge $post (dict "dateModified" (.Lastmod.Format "2006-01-02")) }}{{ end -}} {{- end -}} {{- with .Section -}} {{- $section := site.GetPage . -}} {{- if and $section (ne $.RelPermalink $section.RelPermalink) -}} {{- end -}} {{- end -}} {{- end -}}
